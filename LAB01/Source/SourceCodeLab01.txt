int hour = 0;
int minute = 0;
int second = 0;

while (1)
{
    clearAllClock();

    // Kim giờ
    setNumberOnClock(hour);

    // Kim phút
    setNumberOnClock(minute / 5);

    // Kim giây
    setNumberOnClock(second / 5);

    HAL_Delay(1000);

    second++;

    if (second >= 60)
    {
        second = 0;
        minute++;

        if (minute >= 60)
        {
            minute = 0;
            hour++;

            if (hour >= 12)
            {
                hour = 0;
            }
        }
    }
}
